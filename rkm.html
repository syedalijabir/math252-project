load_rkm_cached_results <- function(embedding_path) {
  embeddings <- read_csv(embedding_path, show_col_types = FALSE)

  user_ids <- embeddings[[1]]
  embedding_matrix <- embeddings[, -1]
  embedding_matrix <- scale(as.matrix(embedding_matrix))

  embedding_tag <- tools::file_path_sans_ext(basename(embedding_path))

  full_file <- file.path(
    "output/rkm_cache",
    paste0("rkm_full_result_", embedding_tag, ".rds")
  )

  if (file.exists(full_file)) {
    return(readRDS(full_file))
  }

  dmat <- dist(embedding_matrix)

  results_grid <- expand.grid(k = k_range, q = q_range)
  results_grid$Silhouette <- NA_real_
  results_grid$CH <- NA_real_
  
  results_grid$runtime_sec <- NA_real_
  results_grid$status <- "ok"
  results_grid$notes <- NA_character_

  rkm_models <- vector("list", nrow(results_grid))

  for (i in seq_len(nrow(results_grid))) {
    k_val <- results_grid$k[i]
    q_val <- results_grid$q[i]

    cache_file <- file.path(
      "output/rkm_cache",
      paste0("rkm_", embedding_tag, "_k", k_val, "_q", q_val, ".rds")
    )

    if (!file.exists(cache_file)) next

    cached <- readRDS(cache_file)

    results_grid$Silhouette[i] <- cached$Silhouette
    results_grid$CH[i] <- cached$CH
    rkm_models[[i]] <- cached$model
  }

  valid_rows <- which(!is.na(results_grid$Silhouette))
  if (length(valid_rows) == 0) return(NULL)

  best_index <- which.max(results_grid$Silhouette)
  best_params <- results_grid[best_index, ]

  best_k <- best_params$k
  best_q <- best_params$q
  best_silhouette <- best_params$Silhouette
  best_rkm <- rkm_models[[best_index]]

  cluster_df <- data.frame(
    UserID = user_ids,
    Cluster = best_rkm$cluster
  )

  best_sil <- silhouette(best_rkm$cluster, dmat)

  pca <- prcomp(embedding_matrix)
  var_explained <- summary(pca)$importance[2, 1:2]

  pca_df <- data.frame(
    PC1 = pca$x[, 1],
    PC2 = pca$x[, 2],
    Cluster = as.factor(best_rkm$cluster)
  )

  list(
    embedding_path = embedding_path,
    embedding_tag = embedding_tag,
    embedding_matrix = embedding_matrix,
    user_ids = user_ids,
    results_grid = results_grid,
    best_index = best_index,
    best_params = best_params,
    best_k = best_k,
    best_q = best_q,
    best_silhouette = best_silhouette,
    best_rkm = best_rkm,
    best_sil = best_sil,
    cluster_df = cluster_df,
    dmat = dmat,
    pca_df = pca_df,
    var_explained = var_explained
  )
}

4. Fit embeddings with caching

all_results <- list()

for (embedding_path in embedding_paths) {

  if (!file.exists(embedding_path)) {
    cat("Missing file:", embedding_path, "\n")
    next
  }
  
  cat("\nProcessing:", embedding_path, "\n")
  
  embedding_t0 <- Sys.time()
  
  embeddings <- read_csv(embedding_path)
  
  user_ids <- embeddings[[1]]
  embedding_matrix <- embeddings[, -1]
  # embedding_matrix <- as.matrix(embedding_matrix)
  embedding_matrix <- scale(embedding_matrix)
  
  embedding_tag <- tools::file_path_sans_ext(basename(embedding_path))
  
  full_file <- file.path(
    "output/rkm_cache",
    paste0("rkm_full_result_", embedding_tag, ".rds")
  )

  if (file.exists(full_file)) {
    cat("Loading FULL cached result for:", embedding_tag, "\n")
    all_results[[embedding_tag]] <- readRDS(full_file)
    next
  }
  
  dmat <- dist(embedding_matrix)
  
  results_grid <- expand.grid(k = k_range, q = q_range)
  results_grid$Silhouette <- NA_real_
  results_grid$CH <- NA_real_
  
  results_grid$runtime_sec <- NA_real_
  results_grid$status <- rep("ok", nrow(results_grid))
  results_grid$notes <- rep(NA_character_, nrow(results_grid))

  
  rkm_models <- vector("list", nrow(results_grid))
  
  for (i in seq_len(nrow(results_grid))) {
    k_val <- results_grid$k[i]
    q_val <- results_grid$q[i]
    
    cache_file <- file.path(
      "output/rkm_cache",
      paste0("rkm_", embedding_tag, "_k", k_val, "_q", q_val, ".rds")
    )
    
    if (file.exists(cache_file)) {
      cat("Loading cached result for", embedding_tag, "k =", k_val, "q =", q_val, "\n")
      cached <- readRDS(cache_file)
      
      model <- cached$model
      avg_sil <- cached$Silhouette
      ch_val <- cached$CH
      
    } else {
      cat("Running RKM for", embedding_tag, "k =", k_val, "q =", q_val, "\n")
      
      t0 <- Sys.time()

      model <- tryCatch(
        cluspca(
          embedding_matrix,
          nclus = k_val,
          ndim = q_val,
          method = "RKM",
          rotation = "varimax"
        ),
        warning = function(w) {
          results_grid$status[i] <- "warning"
          results_grid$notes[i] <- conditionMessage(w)
          invokeRestart("muffleWarning")
        },
        error = function(e) {
          results_grid$status[i] <- "error"
          results_grid$notes[i] <- conditionMessage(e)
          NULL
        }
      )
      
      results_grid$runtime_sec[i] <- as.numeric(difftime(Sys.time(), t0, units = "secs"))
      
      if (is.null(model)) next

      sil <- silhouette(model$cluster, dmat)
      avg_sil <- mean(sil[, 3])
      
      ch_val <- intCriteria(
        traj = embedding_matrix,
        part = as.integer(model$cluster),
        crit = "Calinski_Harabasz"
      )$calinski_harabasz
      
      saveRDS(
        list(
          model = model,
          Silhouette = avg_sil,
          CH = ch_val,
          runtime_sec = results_grid$runtime_sec[i],
          status = results_grid$status[i],
          notes = results_grid$notes[i],
          k = k_val,
          q = q_val,
          embedding_path = embedding_path
        ),
        cache_file
      )
    }
    
    results_grid$Silhouette[i] <- avg_sil
    results_grid$CH[i] <- ch_val
    rkm_models[[i]] <- model
  }
  
  best_index <- which.max(results_grid$Silhouette)
  best_params <- results_grid[best_index, ]

  best_k <- best_params$k
  best_q <- best_params$q
  best_silhouette <- best_params$Silhouette
  best_rkm <- rkm_models[[best_index]]

  cluster_df <- data.frame(
    UserID = user_ids,
    Cluster = best_rkm$cluster
  )
  
  best_sil <- silhouette(best_rkm$cluster, dmat)

  pca <- prcomp(embedding_matrix)
  var_explained <- summary(pca)$importance[2, 1:2]

  pca_df <- data.frame(
    PC1 = pca$x[, 1],
    PC2 = pca$x[, 2],
    Cluster = as.factor(best_rkm$cluster)
  )

  runtime_total_sec <- as.numeric(difftime(Sys.time(), embedding_t0, units = "secs"))
  
  cluster_sizes <- table(cluster_df$Cluster)

  notes_text <- NA_character_
  if ("notes" %in% names(results_grid)) {
    tmp_notes <- unique(na.omit(results_grid$notes))
    if (length(tmp_notes) > 0) {
      notes_text <- paste(tmp_notes, collapse = " | ")
    }
  }
  
  summary_row <- data.frame(
    method = "RKM",
    embedding = embedding_tag,
    best_k = best_k,
    best_q = best_q,
    silhouette = best_params$Silhouette,
    CH = best_params$CH,
    prob_silh = NA,
    cluster_size_min = min(cluster_sizes),
    cluster_size_max = max(cluster_sizes),
    cluster_size_sd = sd(as.numeric(cluster_sizes)),
    runtime_total_sec = runtime_total_sec,
    notes = notes_text
  )
  all_results[[embedding_tag]] <- list(
    embedding_path = embedding_path,
    embedding_tag = embedding_tag,
    embedding_matrix = embedding_matrix,
    user_ids = user_ids,
    results_grid = results_grid,
    best_index = best_index,
    best_params = best_params,
    best_k = best_k,
    best_q = best_q,
    best_silhouette = best_silhouette,
    best_rkm = best_rkm,
    best_sil = best_sil,
    cluster_df = cluster_df,
    dmat = dmat,
    pca_df = pca_df,
    var_explained = var_explained,
    runtime_total_sec = runtime_total_sec,
    summary_row = summary_row,
    stability_info = list(
      seeds_tested = NA,
      n_runs = NA,
      chosen_k_values = NA,
      chosen_q_values = NA,
      ari_mean = NA,
      ari_sd = NA,
      stability_notes = NA_character_
    )
  )
  
  cat("Saved results for:", embedding_tag, "\n")

}
