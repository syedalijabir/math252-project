load_pd_cached_results <- function(embedding_path) {
  embeddings <- read_csv(embedding_path, show_col_types = FALSE)

  user_ids <- embeddings[[1]]
  embedding_matrix <- embeddings[, -1]
  embedding_matrix <- scale(as.matrix(embedding_matrix))

  embedding_tag <- tools::file_path_sans_ext(basename(embedding_path))

  full_file <- file.path(
    "output/pd_cache",
    paste0("pd_full_result_", embedding_tag, ".rds")
  )

  if (file.exists(full_file)) {
    return(readRDS(full_file))
  }

  dmat <- dist(embedding_matrix)

  results <- vector("list", length = max(k_vals))
  avg_silhouette <- rep(NA_real_, max(k_vals))
  ch_scores <- rep(NA_real_, max(k_vals))
  prob_silh_scores <- rep(NA_real_, max(k_vals))

  for (k in k_vals) {
    cache_file <- file.path(
      "output/pd_cache",
      paste0("pd_", embedding_tag, "_k", k, ".rds")
    )

    if (!file.exists(cache_file)) next

    cached <- readRDS(cache_file)

    results[[k]] <- list(
      model = cached$model,
      labels = cached$labels,
      silhouette = cached$silhouette,
      avg_silhouette = cached$avg_silhouette,
      ch = cached$ch,
      prob_silh = cached$prob_silh
    )

    avg_silhouette[k] <- cached$avg_silhouette
    ch_scores[k] <- cached$ch
    prob_silh_scores[k] <- cached$prob_silh
  }

  valid_k <- k_vals[!is.na(avg_silhouette[k_vals])]
  if (length(valid_k) == 0) return(NULL)

  pd_results <- data.frame(
    Method = "PD",
    k = k_vals,
    Silhouette = avg_silhouette[k_vals],
    CH = ch_scores[k_vals],
    ProbSilh = prob_silh_scores[k_vals]
  )

  best_k <- k_vals[which.max(avg_silhouette[k_vals])]
  best_params <- pd_results[pd_results$k == best_k, ]

  best_model <- results[[best_k]]$model
  best_labels <- results[[best_k]]$labels
  best_sil <- results[[best_k]]$silhouette
  final_avg_sil <- results[[best_k]]$avg_silhouette
  best_prob_silh <- results[[best_k]]$prob_silh

  cluster_df <- data.frame(
    UserID = user_ids,
    Cluster = best_labels
  )

  pca <- prcomp(embedding_matrix)
  var_explained <- summary(pca)$importance[2, 1:2]

  pca_df <- data.frame(
    PC1 = pca$x[, 1],
    PC2 = pca$x[, 2],
    Cluster = as.factor(best_labels)
  )

  list(
    embedding_path = embedding_path,
    embedding_tag = embedding_tag,
    embedding_matrix = embedding_matrix,
    user_ids = user_ids,
    pd_results = pd_results,
    results = results,
    best_k = best_k,
    best_params = best_params,
    best_model = best_model,
    best_labels = best_labels,
    best_sil = best_sil,
    best_prob_silh = best_prob_silh,
    final_avg_sil = final_avg_sil,
    cluster_df = cluster_df,
    dmat = dmat,
    pca_df = pca_df,
    var_explained = var_explained
  )
}

3. Evaluate PD across k

# PD: CHOOSE NUMBER OF CLUSTERS
# We evaluate k = 2 to 20 using:
# - average silhouette width
# - Calinski-Harabasz (CH) index
# - probability silhouette

evaluate_pd <- function(x, k_vals, dmat, embedding_path, embedding_tag) {
  runtime_sec <- rep(NA_real_, max(k_vals))
  status <- rep("ok", max(k_vals))
  notes <- rep(NA_character_, max(k_vals))

  results <- vector("list", length = max(k_vals))
  avg_silhouette <- rep(NA_real_, max(k_vals))
  ch_scores <- rep(NA_real_, max(k_vals))
  prob_silh_scores <- rep(NA_real_, max(k_vals))

  for (k in k_vals) {
    cache_file <- file.path(
      "output/pd_cache",
      paste0("pd_", embedding_tag, "_k", k, ".rds")
    )

    if (file.exists(cache_file)) {
      cat("Loading cached result for", embedding_tag, "k =", k, "\n")
      cached <- readRDS(cache_file)

      model <- cached$model
      labels <- cached$labels
      sil <- cached$silhouette
      avg_sil <- cached$avg_silhouette
      ch_val <- cached$ch
      prob_silh <- cached$prob_silh
    } else {
      cat("Running PDC for", embedding_tag, "k =", k, "\n")

      t0 <- Sys.time()
      model <- tryCatch(
      PDC(x, k = k),
      warning = function(w) {
        status[k] <<- "warning"
        notes[k] <<- conditionMessage(w)
        invokeRestart("muffleWarning")
      },
      error = function(e) {
        status[k] <<- "error"
        notes[k] <<- conditionMessage(e)
        NULL
      }
      )
      runtime_sec[k] <- as.numeric(difftime(Sys.time(), t0, units = "secs"))
      if (is.null(model)) next
      
      labels <- model$label
      sil <- silhouette(labels, dmat)
      avg_sil <- mean(sil[, 3])

      ch_val <- intCriteria(
        traj = x,
        part = as.integer(labels),
        crit = "Calinski_Harabasz"
      )$calinski_harabasz

      prob_silh <- Silh(model$probability)

      saveRDS(
        list(
          model = model,
          labels = labels,
          silhouette = sil,
          avg_silhouette = avg_sil,
          ch = ch_val,
          prob_silh = prob_silh,
          k = k,
          embedding_path = embedding_path
        ),
        cache_file
      )
    }

    avg_silhouette[k] <- avg_sil
    ch_scores[k] <- ch_val
    prob_silh_scores[k] <- prob_silh

    results[[k]] <- list(
      model = model,
      labels = labels,
      silhouette = sil,
      avg_silhouette = avg_sil,
      ch = ch_val,
      prob_silh = prob_silh
    )
  }

  pd_results <- data.frame(
    Method = "PD",
    k = k_vals,
    Silhouette = avg_silhouette[k_vals],
    CH = ch_scores[k_vals],
    ProbSilh = prob_silh_scores[k_vals],
    runtime_sec = runtime_sec[k_vals],
    status = status[k_vals],
    notes = notes[k_vals]
  )
  
  saveRDS(
  list(
    model = model,
    labels = labels,
    silhouette = sil,
    avg_silhouette = avg_sil,
    ch = ch_val,
    prob_silh = prob_silh,
    runtime_sec = runtime_sec[k],
    status = status[k],
    notes = notes[k],
    k = k,
    embedding_path = embedding_path
  ),
  cache_file
)

  list(
    results = results,
    pd_results = pd_results
  )
}

5. Loop over all embeddings

for (embedding_path in embedding_paths) {

  if (!file.exists(embedding_path)) {
    cat("Missing file:", embedding_path, "\n")
    next
  }

  cat("\nProcessing:", embedding_path, "\n")

  embedding_t0 <- Sys.time()

  embeddings <- read_csv(embedding_path, show_col_types = FALSE)

  user_ids <- embeddings[[1]]
  embedding_matrix <- embeddings[, -1]
  embedding_matrix <- scale(as.matrix(embedding_matrix))

  embedding_tag <- tools::file_path_sans_ext(basename(embedding_path))

  full_file <- file.path("output/pd_cache", paste0("pd_full_result_", embedding_tag, ".rds"))

  if (file.exists(full_file)) {
    cat("Loading FULL cached result for:", embedding_tag, "\n")
    all_results[[embedding_tag]] <- readRDS(full_file)
    next
  }

  dmat <- dist(embedding_matrix)

  eval_res <- evaluate_pd(
    x = embedding_matrix,
    k_vals = k_vals,
    dmat = dmat,
    embedding_path = embedding_path,
    embedding_tag = embedding_tag
  )

  results <- eval_res$results
  pd_results <- eval_res$pd_results

  best_k <- pd_results$k[which.max(pd_results$Silhouette)]
  best_params <- pd_results[pd_results$k == best_k, ]

  best_model <- results[[best_k]]$model
  best_labels <- results[[best_k]]$labels
  best_sil <- results[[best_k]]$silhouette
  best_prob_silh <- results[[best_k]]$prob_silh
  final_avg_sil <- results[[best_k]]$avg_silhouette

  cluster_df <- data.frame(
    UserID = user_ids,
    Cluster = best_labels
  )

  pca <- prcomp(embedding_matrix)
  var_explained <- summary(pca)$importance[2, 1:2]

  pca_df <- data.frame(
    PC1 = pca$x[, 1],
    PC2 = pca$x[, 2],
    Cluster = as.factor(best_labels)
  )

  runtime_total_sec <- as.numeric(difftime(Sys.time(), embedding_t0, units = "secs"))

  cluster_sizes <- table(cluster_df$Cluster)

  notes_text <- NA_character_
  if ("notes" %in% names(pd_results)) {
    tmp_notes <- unique(na.omit(pd_results$notes))
    if (length(tmp_notes) > 0) {
      notes_text <- paste(tmp_notes, collapse = " | ")
    }
  }

  cluster_sizes <- table(cluster_df$Cluster)

  summary_row <- data.frame(
    method = "PD",
    embedding = embedding_tag,
    best_k = best_k,
    best_q = NA,
    silhouette = final_avg_sil,
    CH = best_params$CH,
    prob_silh = best_params$ProbSilh,
    cluster_size_min = min(cluster_sizes),
    cluster_size_max = max(cluster_sizes),
    cluster_size_sd = sd(as.numeric(cluster_sizes)),
    runtime_total_sec = runtime_total_sec,
    notes = paste(unique(na.omit(pd_results$notes)), collapse = " | ")
  )

  all_results[[embedding_tag]] <- list(
    embedding_path = embedding_path,
    embedding_tag = embedding_tag,
    embedding_matrix = embedding_matrix,
    user_ids = user_ids,
    pd_results = pd_results,
    results = results,
    best_k = best_k,
    best_params = best_params,
    best_model = best_model,
    best_labels = best_labels,
    best_sil = best_sil,
    best_prob_silh = best_prob_silh,
    final_avg_sil = final_avg_sil,
    cluster_df = cluster_df,
    dmat = dmat,
    pca_df = pca_df,
    var_explained = var_explained,
    runtime_total_sec = runtime_total_sec,
    summary_row = summary_row,
    stability_info = list(
      seeds_tested = NA,
      n_runs = NA,
      chosen_k_values = NA,
      chosen_q_values = NA,
      ari_mean = NA,
      ari_sd = NA,
      stability_notes = NA_character_
    )
  )

  cat("Saved results for:", embedding_tag, "\n")
}
